#create_html_css_recording(html, params = {}) ⇒ HTMLCSSToImage::ApiResponse

Records an HTML document or fragment with optional CSS and Google Fonts. Returns the recording ID, status URL, and media URL while rendering can continue in the background. Optional settings use the API defaults.

Parameters:

  • html (String) —

    HTML to record, as a fragment or a complete document

  • params (Hash) (defaults to: {}) —

    browser, timeline, and output options passed to the API

Options Hash (params):

  • :duration_ms (Integer) —

    Required recording duration in milliseconds: 500–30000, or up to 90000 in target mode.

  • :css (String) —

    CSS used to style the recording.

  • :google_fonts (String) —

    Google Fonts to load, separated by |. Set font-family in the CSS to use them.

  • :device_scale (Numeric) —

    Capture pixel density. HTML/CSS defaults to 2; URL recordings default to 1.

  • :viewport_height (Integer) —

    Chrome viewport height in CSS pixels. Supply both viewport dimensions to set the recording size.

  • :viewport_width (Integer) —

    Chrome viewport width in CSS pixels. Supply both viewport dimensions to set the recording size.

  • :disable_twemoji (Boolean) —

    True disables Twemoji for HTML/CSS. URL recordings inject Twemoji only when explicitly false.

  • :dedupe_duration_s (Integer) —

    Reuse an identical recent recording for this many seconds. Set to 0 to disable deduplication.

  • :color_scheme (String) —

    Chrome's preferred color scheme: light or dark.

  • :timezone (String) —

    Browser timezone as an IANA identifier, such as America/New_York.

  • :viewport_mobile (Boolean) —

    Enable mobile viewport behavior, including the page's viewport meta tag.

  • :viewport_landscape (Boolean) —

    Emulate landscape orientation.

  • :viewport_touch (Boolean) —

    Emulate touch support.

  • :media_type (String) —

    CSS media type used while recording: print or screen.

  • :proxy_id (String) —

    Configured organization proxy to use while recording.

  • :storage_destination_id (String) —

    Configured storage destination to receive the recording.

  • :request_overrides (Array<Hash>) —

    Browser request block rules. Requires a paid plan.

  • :trigger (Hash) —

    Start condition: type: "immediate" (default), "element_visible" with a selector, or "function_call" to wait for window.RecordingReady(). Waiting triggers accept timeout_ms.

  • :actions (Array<Hash>) —

    Ordered timeline interactions: simple_scroll, scroll, stall, click, locate_selector, execute_javascript, or error. Each action has a type discriminator. Omit or use an empty array to record without interactions.

  • :output (Hash) —

    Media format (mp4, webm, or gif), fps, and pixel scale. Defaults to MP4 at 30 fps; GIF defaults to 15 fps. Recordings have no audio.

  • :duration_mode (String) —

    at_most (default) stops when actions finish or the duration is reached. fixed pads or cuts off at that duration. at_least lets actions finish up to 90 seconds. target adjusts playback speed to fit the duration.

  • :defer_render (Boolean) —

    Wait to render until media or storage is requested. Defaults to false; does not reset deduplicated recordings.

  • :skip_scroll_preload (Boolean) —

    Skip the preparation scroll that loads lazy content. Scrolling actions still run. Defaults to false.

Returns:

  • (HTMLCSSToImage::ApiResponse) —

    recording details at .id, .status_url, and .recording_url

See Also:

  • https://docs.htmlcsstoimage.com/getting-started/recordings/
  • https://docs.htmlcsstoimage.com/parameters/actions/
  • https://docs.htmlcsstoimage.com/parameters/trigger/
  • https://docs.htmlcsstoimage.com/parameters/output/

def create_image_from_template(
  template_id,
  template_values = {},
  params = {},
  template_version: nil,
  format: nil,
  **keyword_values
)
  params ||= {}
  template_values = template_values.merge(keyword_values)
  legacy_version =
    if params.is_a?(Hash)
      params[:template_version] || params["template_version"]
    end

  options = { template_version: template_version || legacy_version }
  options[:format] = format unless format.nil?

  generate_templated_image_url(template_id, template_values, **options)
end

def create_templated_image(
  template_id,
  template_values = {},
  template_version: nil,
  format: nil,
  **keyword_values
)
  template_values = template_values.merge(keyword_values)
  body = {
    template_id: template_id,
    template_values: template_values
  }
  body[:template_version] = template_version unless template_version.nil?
  body[:format] = format unless format.nil?

  self.class.post(
    "/v1/image",
    basic_auth: @auth,
    body: body.to_json,
    query: { includeId: true }
  )
end

def generate_create_and_render_url(url, params = {})
  pairs = [["url", url.to_s]]
  format = params[:format] || params["format"]

  params
    .reject do |key, _value|
      %w[url format pdf_options dedupe_duration_s request_overrides].include?(key.to_s)
    end
    .sort_by { |key, _value| key.to_s }
    .each do |key, value|
      next if value.nil?
      next if value == false && key.to_s != "transparent_background"

      case key.to_s
      when "headers"
        value.each { |name, header_value| pairs << ["headers", "#{name}:#{header_value}"] }
      when "additional_header_origins"
        value.each { |origin| pairs << ["additional_header_origins", origin.to_s] }
      else
        pairs << [key.to_s, signed_value(value)]
      end
    end

  query = Addressable::URI.form_encode(pairs)
  token = generate_hmac_token(query)
  format_path = format.nil? ? "" : "/#{format}"

  ApiResponse.new(
    url: "https://hcti.io/v1/image/create-and-render/#{@auth[:username]}/#{token}#{format_path}?#{query}"
  )
end

def generate_templated_image_url(
  template_id,
  template_values = {},
  template_version: nil,
  format: nil,
  **keyword_values
)
  template_values = template_values.merge(keyword_values)
  pairs = []
  pairs << ["template_version", template_version.to_s] unless template_version.nil?

  template_values.sort_by { |key, _value| key.to_s }.each do |key, value|
    next if value.nil?

    pairs << [key.to_s, signed_value(value)]
  end

  query = Addressable::URI.form_encode(pairs)
  token = generate_hmac_token(query)
  separator = query.empty? ? "" : "?"
  format_path = format.nil? ? "" : "/#{format}"

  ApiResponse.new(
    url: "https://hcti.io/v1/image/#{template_id}/#{token}#{format_path}#{separator}#{query}"
  )
